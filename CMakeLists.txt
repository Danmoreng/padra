cmake_minimum_required(VERSION 3.28)
project(padra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT MSVC)
  # Match the measured host runtime configuration; CUDA kernels remain -O3.
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-O2>")
endif()
option(PADRA_ENABLE_CUDA "Build the native CUDA inference runtime" ON)
option(PADRA_ENABLE_NATIVE_TEXT "Build the locked native tokenizer and renderer" ON)
option(PADRA_EXL3_PREFILL_ASYNC64 "Enable the experimental exact EXL3 K4 K64 prefill transport" OFF)
find_package(Threads REQUIRED)
add_library(padra_core
  src/util/json.cpp src/model/safetensors.cpp src/numeric/nvfp4.cpp
  src/compiler/sha256.cpp src/model/compiled.cpp src/runtime/inference.cpp
  src/runtime/model_runtime.cpp src/text/tokenizer.cpp src/text/stream.cpp)
target_include_directories(padra_core PUBLIC include src)
target_link_libraries(padra_core PRIVATE Threads::Threads)
target_compile_definitions(padra_core PUBLIC MELLUM16_HAS_CUDA=$<BOOL:${PADRA_ENABLE_CUDA}>)
target_compile_definitions(padra_core PRIVATE MELLUM16_NATIVE_TEXT=$<BOOL:${PADRA_ENABLE_NATIVE_TEXT}>)
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
  target_link_libraries(padra_core PRIVATE OpenSSL::Crypto)
  target_compile_definitions(padra_core PRIVATE MELLUM16_HAS_OPENSSL=1)
endif()
if(WIN32)
  target_sources(padra_core PRIVATE src/platform/mapped_file_windows.cpp)
else()
  target_sources(padra_core PRIVATE src/platform/mapped_file_posix.cpp)
endif()
if(MSVC)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
else()
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
endif()
set(PADRA_BUILD_SIMT 0)
set(PADRA_BUILD_ARCHITECTURE "none")
if(PADRA_ENABLE_CUDA)
  include(cmake/PadraCuda.cmake)
  padra_add_cuda()
endif()
configure_file(src/runtime/gpu_build_config.h.in generated/runtime/gpu_build_config.h @ONLY)
target_include_directories(padra_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(PADRA_ENABLE_NATIVE_TEXT)
  include(cmake/PadraNativeText.cmake)
  padra_add_native_text()
endif()
include(cmake/PadraHttp.cmake)
padra_add_http()
add_library(padra_server
  src/server/chat_request.cpp src/server/coordinator.cpp src/server/secure_id.cpp
  src/server/session_store.cpp src/server/tool_protocol.cpp src/server/responses.cpp
  src/server/response_store.cpp src/server/response_stream.cpp src/server/http_server.cpp)
target_link_libraries(padra_server PUBLIC padra_core padra_http Threads::Threads)
if(WIN32)
  target_link_libraries(padra_server PRIVATE bcrypt)
endif()
add_executable(padra-server src/cli/server_main.cpp)
target_link_libraries(padra-server PRIVATE padra_server)
option(PADRA_BUILD_DIAGNOSTICS "Build explicit offline SIMT qualification tools" OFF)
if(PADRA_BUILD_DIAGNOSTICS AND PADRA_ENABLE_CUDA AND PADRA_BUILD_SIMT)
  add_executable(padra-simt-qualify tools/simt_qualify.cpp)
  target_link_libraries(padra-simt-qualify PRIVATE padra_core)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--ptxas-options=-v>")
endif()
option(PADRA_BUILD_TESTS "Build checkpoint-free host and CUDA regression tests" OFF)
if(PADRA_BUILD_TESTS)
  enable_testing()
  add_executable(padra-profile-test tests/gpu_profiles.cpp)
  target_link_libraries(padra-profile-test PRIVATE padra_core)
  add_test(NAME gpu-profiles COMMAND padra-profile-test)
  if(PADRA_ENABLE_CUDA)
    add_executable(padra-simt-kernel-test tests/simt_kernels.cu)
    target_include_directories(padra-simt-kernel-test PRIVATE include src "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(padra-simt-kernel-test PRIVATE CUDA::cudart)
    set_target_properties(padra-simt-kernel-test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES "${architecture}")
    target_compile_options(padra-simt-kernel-test PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3>")
    add_test(NAME simt-kernels COMMAND padra-simt-kernel-test)
    set_tests_properties(simt-kernels PROPERTIES TIMEOUT 120)
    add_executable(padra-simt-operator-test tests/simt_operators.cu)
    target_include_directories(padra-simt-operator-test PRIVATE include src "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(padra-simt-operator-test PRIVATE CUDA::cudart)
    set_target_properties(padra-simt-operator-test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES "${architecture}")
    target_compile_options(padra-simt-operator-test PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>")
    add_test(NAME simt-operators COMMAND padra-simt-operator-test)
    set_tests_properties(simt-operators PROPERTIES TIMEOUT 120)
    add_executable(padra-simt-prefill-test tests/simt_prefill.cu)
    target_include_directories(padra-simt-prefill-test PRIVATE include src "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(padra-simt-prefill-test PRIVATE CUDA::cudart)
    set_target_properties(padra-simt-prefill-test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES "${architecture}")
    target_compile_options(padra-simt-prefill-test PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>")
    add_test(NAME simt-prefill COMMAND padra-simt-prefill-test)
    set_tests_properties(simt-prefill PROPERTIES TIMEOUT 180)
  endif()
endif()
