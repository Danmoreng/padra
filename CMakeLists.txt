# Fingerprint the actual sources at configure time, including an edited tree.
file(GLOB_RECURSE padra_identity_files CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/*" "${PROJECT_SOURCE_DIR}/include/*"
  "${PROJECT_SOURCE_DIR}/cmake/*" "${PROJECT_SOURCE_DIR}/third_party/*")
list(APPEND padra_identity_files "${PROJECT_SOURCE_DIR}/CMakeLists.txt")
list(SORT padra_identity_files)
set(padra_identity_material "")
foreach(source IN LISTS padra_identity_files)
  file(SHA256 "${source}" source_hash)
  file(RELATIVE_PATH source_name "${PROJECT_SOURCE_DIR}" "${source}")
  string(APPEND padra_identity_material "${source_name}:${source_hash}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
endforeach()
string(SHA256 PADRA_BUILD_SOURCE_SHA256 "${padra_identity_material}")

function(padra_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
  else()
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
  endif()
endfunction()
