cmake_minimum_required(VERSION 3.28)
project(padra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT MSVC)
  # Match the measured host runtime configuration; CUDA kernels remain -O3.
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-O2>")
endif()
option(PADRA_ENABLE_CUDA "Build the native SM120 inference runtime" ON)
option(PADRA_ENABLE_NATIVE_TEXT "Build the locked native tokenizer and renderer" ON)
find_package(Threads REQUIRED)
add_library(padra_core
  src/util/json.cpp src/model/safetensors.cpp src/numeric/nvfp4.cpp
  src/compiler/sha256.cpp src/model/compiled.cpp src/runtime/inference.cpp
  src/runtime/model_runtime.cpp src/text/tokenizer.cpp src/text/stream.cpp)
target_include_directories(padra_core PUBLIC include src)
target_link_libraries(padra_core PRIVATE Threads::Threads)
target_compile_definitions(padra_core PUBLIC MELLUM16_HAS_CUDA=$<BOOL:${PADRA_ENABLE_CUDA}>)
target_compile_definitions(padra_core PRIVATE MELLUM16_NATIVE_TEXT=$<BOOL:${PADRA_ENABLE_NATIVE_TEXT}>)
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
  target_link_libraries(padra_core PRIVATE OpenSSL::Crypto)
  target_compile_definitions(padra_core PRIVATE MELLUM16_HAS_OPENSSL=1)
endif()
if(WIN32)
  target_sources(padra_core PRIVATE src/platform/mapped_file_windows.cpp)
else()
  target_sources(padra_core PRIVATE src/platform/mapped_file_posix.cpp)
endif()
if(MSVC)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
else()
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
endif()
if(PADRA_ENABLE_CUDA)
  set(PADRA_CUDA_ARCHITECTURE "120a" CACHE STRING "Qualified native architecture: 120a or 120")
  if(NOT PADRA_CUDA_ARCHITECTURE MATCHES "^120a?$")
    message(FATAL_ERROR "Only SM120/SM120a is currently implemented")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}")
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  include(cmake/PadraCutlass.cmake)
  padra_add_cutlass()
  target_link_libraries(padra_core PRIVATE Padra::CUTLASS)
  target_sources(padra_core PRIVATE src/cuda/inference.cu src/cuda/cutlass_fp8.cu)
  target_link_libraries(padra_core PRIVATE CUDA::cudart)
  target_compile_definitions(padra_core PRIVATE MELLUM16_ARCHITECTURE="${PADRA_CUDA_ARCHITECTURE}")
  set_target_properties(padra_core PROPERTIES CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}" CUDA_STANDARD 20)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3>")
endif()
if(PADRA_ENABLE_NATIVE_TEXT)
  include(cmake/PadraNativeText.cmake)
  padra_add_native_text()
endif()
include(cmake/PadraHttp.cmake)
padra_add_http()
add_library(padra_server
  src/server/chat_request.cpp src/server/coordinator.cpp src/server/secure_id.cpp
  src/server/session_store.cpp src/server/tool_protocol.cpp src/server/responses.cpp
  src/server/response_store.cpp src/server/response_stream.cpp src/server/http_server.cpp)
target_link_libraries(padra_server PUBLIC padra_core padra_http Threads::Threads)
if(WIN32)
  target_link_libraries(padra_server PRIVATE bcrypt)
endif()
add_executable(padra-server src/cli/server_main.cpp)
target_link_libraries(padra-server PRIVATE padra_server)
