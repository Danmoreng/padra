cmake_minimum_required(VERSION 3.28)
project(padra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT MSVC)
  # Match the measured host runtime configuration; CUDA kernels remain -O3.
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-O2>")
endif()
option(PADRA_ENABLE_CUDA "Build the native SM120 inference runtime" ON)
option(PADRA_ENABLE_NATIVE_TEXT "Build the locked native tokenizer and renderer" ON)
find_package(Threads REQUIRED)
add_library(padra_core
  src/util/json.cpp src/model/safetensors.cpp src/numeric/nvfp4.cpp
  src/compiler/sha256.cpp src/model/compiled.cpp src/runtime/inference.cpp
  src/runtime/model_runtime.cpp src/text/tokenizer.cpp src/text/stream.cpp)
target_include_directories(padra_core PUBLIC include src)
target_link_libraries(padra_core PRIVATE Threads::Threads)
target_compile_definitions(padra_core PUBLIC MELLUM16_HAS_CUDA=$<BOOL:${PADRA_ENABLE_CUDA}>)
target_compile_definitions(padra_core PRIVATE MELLUM16_NATIVE_TEXT=$<BOOL:${PADRA_ENABLE_NATIVE_TEXT}>)
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
  target_link_libraries(padra_core PRIVATE OpenSSL::Crypto)
  target_compile_definitions(padra_core PRIVATE MELLUM16_HAS_OPENSSL=1)
endif()
if(WIN32)
  target_sources(padra_core PRIVATE src/platform/mapped_file_windows.cpp)
else()
  target_sources(padra_core PRIVATE src/platform/mapped_file_posix.cpp)
endif()
if(MSVC)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
else()
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
endif()
if(PADRA_ENABLE_CUDA)
  set(PADRA_CUDA_ARCHITECTURE "120a" CACHE STRING "Qualified native architecture: 120a or 120")
  if(NOT PADRA_CUDA_ARCHITECTURE MATCHES "^120a?$")
    message(FATAL_ERROR "Only SM120/SM120a is currently implemented")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}")
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(PADRA_CUTLASS_ROOT "$ENV{HOME}/.cache/padra/cutlass/db1c288993354c88e551c40c19a8fb93a774a241" CACHE PATH "Verified pinned CUTLASS cache")
  if(NOT EXISTS "${PADRA_CUTLASS_ROOT}/SOURCE_LOCK.json")
    message(FATAL_ERROR "Prepare the pinned CUTLASS cache with tools/prepare_cutlass.py; set PADRA_CUTLASS_ROOT")
  endif()
  file(READ "${PADRA_CUTLASS_ROOT}/SOURCE_LOCK.json" cutlass_lock)
  file(SHA256 "${PADRA_CUTLASS_ROOT}/SOURCE_LOCK.json" cutlass_lock_hash)
  if(NOT cutlass_lock_hash STREQUAL "c1d04d1dd2c64156a827bc31ce267a3a9023291c8ef7ffbc35c9488c32075557")
    message(FATAL_ERROR "CUTLASS source lock hash mismatch")
  endif()
  string(JSON cutlass_revision GET "${cutlass_lock}" revision)
  if(NOT cutlass_revision STREQUAL "db1c288993354c88e551c40c19a8fb93a774a241")
    message(FATAL_ERROR "CUTLASS source revision mismatch")
  endif()
  string(JSON cutlass_files LENGTH "${cutlass_lock}" sha256)
  math(EXPR cutlass_last "${cutlass_files}-1")
  foreach(index RANGE ${cutlass_last})
    string(JSON cutlass_file MEMBER "${cutlass_lock}" sha256 ${index})
    string(JSON cutlass_expected GET "${cutlass_lock}" sha256 "${cutlass_file}")
    file(SHA256 "${PADRA_CUTLASS_ROOT}/${cutlass_file}" cutlass_actual)
    if(NOT cutlass_actual STREQUAL cutlass_expected)
      message(FATAL_ERROR "CUTLASS header hash mismatch: ${cutlass_file}")
    endif()
  endforeach()
  target_include_directories(padra_core SYSTEM PRIVATE "${PADRA_CUTLASS_ROOT}/include" "${PADRA_CUTLASS_ROOT}/tools/util/include")
  target_sources(padra_core PRIVATE src/cuda/inference.cu src/cuda/cutlass_fp8.cu)
  target_link_libraries(padra_core PRIVATE CUDA::cudart)
  target_compile_definitions(padra_core PRIVATE MELLUM16_ARCHITECTURE="${PADRA_CUDA_ARCHITECTURE}")
  set_target_properties(padra_core PROPERTIES CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}" CUDA_STANDARD 20)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3>")
endif()
if(PADRA_ENABLE_NATIVE_TEXT)
  find_program(PADRA_CARGO_EXECUTABLE cargo REQUIRED)
  set(PADRA_CARGO_HOME "" CACHE PATH "Optional private Cargo cache")
  set(PADRA_RUSTUP_HOME "" CACHE PATH "Optional private Rustup toolchains")
  set(text_build_environment)
  if(PADRA_CARGO_HOME)
    list(APPEND text_build_environment "CARGO_HOME=${PADRA_CARGO_HOME}")
  endif()
  if(PADRA_RUSTUP_HOME)
    list(APPEND text_build_environment "RUSTUP_HOME=${PADRA_RUSTUP_HOME}")
  endif()
  set(text_archive "${CMAKE_CURRENT_BINARY_DIR}/native-text/release/libmellum16_native_text.a")
  add_custom_command(OUTPUT "${text_archive}"
    COMMAND ${CMAKE_COMMAND} -E env ${text_build_environment} ${PADRA_CARGO_EXECUTABLE}
      build --locked --release --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/src/text/native/Cargo.toml"
      --target-dir "${CMAKE_CURRENT_BINARY_DIR}/native-text" -j2
    DEPENDS src/text/native/Cargo.toml src/text/native/Cargo.lock src/text/native/src/lib.rs src/text/native/src/renderer.rs
    VERBATIM)
  add_custom_target(padra-native-text-build DEPENDS "${text_archive}")
  add_library(padra_native_text STATIC IMPORTED GLOBAL)
  set_target_properties(padra_native_text PROPERTIES IMPORTED_LOCATION "${text_archive}")
  add_dependencies(padra_native_text padra-native-text-build)
  target_link_libraries(padra_core PRIVATE padra_native_text ${CMAKE_DL_LIBS} m)
endif()
add_library(padra_http STATIC third_party/cpp-httplib/httplib.cpp)
target_include_directories(padra_http SYSTEM PUBLIC third_party/cpp-httplib)
target_compile_definitions(padra_http PRIVATE CPPHTTPLIB_TCP_NODELAY=1)
target_link_libraries(padra_http PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(padra_http PUBLIC ws2_32)
endif()
if(NOT MSVC)
  target_compile_options(padra_http PRIVATE -w)
endif()
add_library(padra_server
  src/server/chat_request.cpp src/server/coordinator.cpp src/server/secure_id.cpp
  src/server/session_store.cpp src/server/tool_protocol.cpp src/server/responses.cpp
  src/server/response_store.cpp src/server/response_stream.cpp src/server/http_server.cpp)
target_link_libraries(padra_server PUBLIC padra_core padra_http Threads::Threads)
if(WIN32)
  target_link_libraries(padra_server PRIVATE bcrypt)
endif()
add_executable(padra-server src/cli/server_main.cpp)
target_link_libraries(padra-server PRIVATE padra_server)
