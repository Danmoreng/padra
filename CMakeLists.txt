cmake_minimum_required(VERSION 3.28)
project(padra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT MSVC)
  # Match the measured host runtime configuration; CUDA kernels remain -O3.
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-O2>")
endif()
option(PADRA_ENABLE_CUDA "Build the native SM120 inference runtime" ON)
option(PADRA_ENABLE_NATIVE_TEXT "Build the locked native tokenizer and renderer" ON)
option(PADRA_EXL3_PREFILL_ASYNC64 "Legacy alias for the exact-fusions-v1 kernel profile" OFF)
set(PADRA_KERNEL_PROFILE "" CACHE STRING "Versioned kernel profile: baseline-v1 or exact-fusions-v1; empty honors the legacy alias")
set_property(CACHE PADRA_KERNEL_PROFILE PROPERTY STRINGS "" baseline-v1 exact-fusions-v1)
if(PADRA_KERNEL_PROFILE STREQUAL "")
  if(PADRA_EXL3_PREFILL_ASYNC64)
    set(PADRA_ACTIVE_KERNEL_PROFILE "exact-fusions-v1")
  else()
    set(PADRA_ACTIVE_KERNEL_PROFILE "baseline-v1")
  endif()
elseif(PADRA_KERNEL_PROFILE STREQUAL "baseline-v1" OR PADRA_KERNEL_PROFILE STREQUAL "exact-fusions-v1")
  set(PADRA_ACTIVE_KERNEL_PROFILE "${PADRA_KERNEL_PROFILE}")
  if(PADRA_EXL3_PREFILL_ASYNC64 AND PADRA_KERNEL_PROFILE STREQUAL "baseline-v1")
    message(FATAL_ERROR "baseline-v1 conflicts with PADRA_EXL3_PREFILL_ASYNC64=ON")
  endif()
else()
  message(FATAL_ERROR "Unsupported PADRA_KERNEL_PROFILE; use baseline-v1 or exact-fusions-v1")
endif()
set(PADRA_ENABLE_EXACT_FUSIONS OFF)
if(PADRA_ACTIVE_KERNEL_PROFILE STREQUAL "exact-fusions-v1")
  set(PADRA_ENABLE_EXACT_FUSIONS ON)
endif()
# Fingerprint the actual sources at configure time, including an edited tree.
file(GLOB_RECURSE padra_identity_files CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/*" "${PROJECT_SOURCE_DIR}/include/*"
  "${PROJECT_SOURCE_DIR}/cmake/*" "${PROJECT_SOURCE_DIR}/third_party/*")
list(APPEND padra_identity_files "${PROJECT_SOURCE_DIR}/CMakeLists.txt")
list(SORT padra_identity_files)
set(padra_identity_material "")
foreach(source IN LISTS padra_identity_files)
  file(SHA256 "${source}" source_hash)
  file(RELATIVE_PATH source_name "${PROJECT_SOURCE_DIR}" "${source}")
  string(APPEND padra_identity_material "${source_name}:${source_hash}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
endforeach()
string(SHA256 PADRA_BUILD_SOURCE_SHA256 "${padra_identity_material}")
configure_file(cmake/build_identity.h.in generated/runtime/build_identity.h @ONLY)
find_package(Threads REQUIRED)
include(CTest)
option(PADRA_ENABLE_SANITIZERS "Instrument host code with ASan and UBSan" OFF)
if(PADRA_ENABLE_SANITIZERS)
  if(MSVC OR PADRA_ENABLE_CUDA)
    message(FATAL_ERROR "Sanitizers require a GCC/Clang host-only build")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
function(padra_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX>")
  else()
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
  endif()
endfunction()
add_library(padra_core
  src/util/json.cpp src/model/safetensors.cpp src/numeric/nvfp4.cpp
  src/compiler/sha256.cpp src/model/compiled.cpp src/runtime/inference.cpp
  src/runtime/model_runtime.cpp src/text/tokenizer.cpp src/text/decoder.cpp src/text/stream.cpp)
target_include_directories(padra_core PUBLIC include src)
target_link_libraries(padra_core PRIVATE Threads::Threads)
target_compile_definitions(padra_core PUBLIC MELLUM16_HAS_CUDA=$<BOOL:${PADRA_ENABLE_CUDA}>)
target_compile_definitions(padra_core PRIVATE MELLUM16_NATIVE_TEXT=$<BOOL:${PADRA_ENABLE_NATIVE_TEXT}>)
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
  target_link_libraries(padra_core PRIVATE OpenSSL::Crypto)
  target_compile_definitions(padra_core PRIVATE MELLUM16_HAS_OPENSSL=1)
endif()
if(WIN32)
  target_sources(padra_core PRIVATE src/platform/mapped_file_windows.cpp)
else()
  target_sources(padra_core PRIVATE src/platform/mapped_file_posix.cpp)
endif()
padra_warnings(padra_core)
if(PADRA_ENABLE_CUDA)
  set(PADRA_CUDA_ARCHITECTURE "120a" CACHE STRING "Qualified native architecture: 120a or 120")
  if(NOT PADRA_CUDA_ARCHITECTURE MATCHES "^120a?$")
    message(FATAL_ERROR "Only SM120/SM120a is currently implemented")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}")
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  include(cmake/PadraCutlass.cmake)
  padra_add_cutlass()
  target_link_libraries(padra_core PRIVATE Padra::CUTLASS)
  target_sources(padra_core PRIVATE src/cuda/inference.cu src/cuda/cutlass_fp8.cu)
  target_link_libraries(padra_core PRIVATE CUDA::cudart)
  target_compile_definitions(padra_core PRIVATE MELLUM16_ARCHITECTURE="${PADRA_CUDA_ARCHITECTURE}")
  target_compile_definitions(padra_core PRIVATE PADRA_EXL3_PREFILL_ASYNC64=$<BOOL:${PADRA_ENABLE_EXACT_FUSIONS}>)
  set_target_properties(padra_core PROPERTIES CUDA_ARCHITECTURES "${PADRA_CUDA_ARCHITECTURE}" CUDA_STANDARD 20)
  target_compile_options(padra_core PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3>")
endif()
if(PADRA_ENABLE_NATIVE_TEXT)
  include(cmake/PadraNativeText.cmake)
  padra_add_native_text()
endif()
include(cmake/PadraHttp.cmake)
padra_add_http()

  src/server/session_store.cpp src/server/tool_protocol.cpp src/server/responses.cpp
  src/server/response_store.cpp src/server/response_stream.cpp src/server/http_server.cpp)
